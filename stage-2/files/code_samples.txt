# Calculate portfolio returns
def calculate_returns(prices):
    returns = []
    for i in range(1, len(prices)):
        r = (prices[i] - prices[i-1]) / prices[i-1]
        returns.append(r)
    return returns

# Format trade confirmation
def format_confirmation(trade_id, ticker, quantity, price):
    return f"CONFIRMED: {trade_id} | {ticker} | QTY: {quantity} | PRICE: ${price:.2f}"

# Risk threshold check
def check_risk(exposure, limit=500000):
    if exposure > limit:
        return "BREACH: exposure exceeds limit"
    return "OK: within threshold"
